
# Aktifkan sekarang
sysctl -w net.ipv4.ip_forward=1

# Tanam permanen ke system (survive reboot)
echo "net.ipv4.ip_forward = 1" >> /etc/sysctl.conf
sysctl -p

# Verifikasi
cat /proc/sys/net/ipv4/ip_forward   # harus output: 1

wg genkey | tee client_privatekey | wg pubkey > client_publickey

#	 Edit .env — isi 3 nilai ini:
#    WG_CLIENT_PRIVATE_KEY = isi dari client_privatekey
#    WG_SERVER_PUBLIC_KEY  = public key server WireGuard
#    WG_SERVER_ENDPOINT    = IP/domain server
#    WG_CLIENT_IP          = misal 100.10.11.2

# 4. Permission & jalankan
chmod +x entrypoint.sh
docker compose up -d
docker compose logs -f
